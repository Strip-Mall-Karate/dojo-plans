# Each case: `== name`, then `-- setup` (bash run first in both trees, with
# $pkg_prefix set), then `-- body` (a callback body, run by bash in one tree
# and translated to Fennel for the other). The body runs in the source
# directory, as a callback did.

== cp-file-into-dir
-- setup
echo tool > tool
-- body
mkdir -p "$pkg_prefix/bin"
cp tool "$pkg_prefix/bin"

== cp-r-tree
-- setup
mkdir -p share/a/b; echo x > share/a/b/f; echo y > share/top
-- body
cp -r share "$pkg_prefix/share"

== cp-glob-into-dir
-- setup
echo 1 > a.conf; echo 2 > b.conf; echo 3 > c.txt
-- body
mkdir -p "$pkg_prefix/etc"
cp ./*.conf "$pkg_prefix/etc/"

== mv-and-rm
-- setup
echo keep > keep; echo gone > gone; mkdir -p d/e; echo z > d/e/f
-- body
mv keep kept
rm -f gone
rm -rf d
rm -f does-not-exist

== ln-s-and-rs
-- setup
mkdir -p p/bin; echo py > p/bin/python3.7
-- body

ln -rs "$pkg_prefix/bin/python3.7" "$pkg_prefix/bin/python"
ln -sf python3.7 "$pkg_prefix/bin/py"

== gcc-hand-ports
-- setup
mkdir -p gcc/config/i386 gcc/config/arm .hidden sub
printf '#define GLIBC_DYNAMIC_LINKER "/lib64/ld-linux-x86-64.so.2"\n#define UCLIBC_DYNAMIC_LINKER32 "/lib/ld-uClibc.so.0"\nother\n' > gcc/config/i386/linux64.h
printf '#define GLIBC_DYNAMIC_LINKER "/lib/ld-linux.so.3"\n' > gcc/config/linux-gnu.h
printf 'nothing here\n' > gcc/config/arm/plain.h
printf 'run /usr/bin/file x\n' > sub/script; printf 'run /usr/bin/file x\n' > .hidden/script; printf 'caf\xe9 /usr/bin/file\n' > latin1
mkdir -p bin; echo gcc-bin > bin/gcc; echo gcc-bin > bin/x86_64-gcc; echo other > bin/x86_64-gcc-ar
echo gxx-bin > bin/g++; echo gxx-bin > bin/c++; echo gxx-bin > bin/x86_64-g++
-- body
local glibc="/opt/glibc%1"
for header in "gcc/config/"*-gnu.h "gcc/config/"*"/"*.h; do
    grep -q LIBC_DYNAMIC_LINKER "$header" || continue
    build_line "  Fixing $header"
    sed -i "$header" \
      -e 's|define[[:blank:]]*\([UCG]\+\)LIBC_DYNAMIC_LINKER\([0-9]*\)[[:blank:]]"\([^\"]\+\)"$|define \1LIBC_DYNAMIC_LINKER\2 "'"${glibc}"'\3"|g' \
      -e 's|/lib64/ld-linux-|/lib/ld-linux-|g'
  done
grep -lr /usr/bin/file ./* | while read -r f; do
    sed -i -e "s,/usr/bin/file,file,g" "$f"
  done
mkdir -p "$pkg_prefix/bin"
cp -a bin/. "$pkg_prefix/bin/"
for bin in "$pkg_prefix/bin/"*-gcc*; do
      if cmp -s "$pkg_prefix/bin/gcc" "$bin"; then
        ln -sfnv gcc "$bin"
      fi
    done
for bin in "$pkg_prefix/bin/c++" "$pkg_prefix/bin/"*-c++* "$pkg_prefix/bin/"*-g++*; do
      if cmp -s "$pkg_prefix/bin/g++" "$bin"; then
        ln -sfn g++ "$bin"
      fi
    done

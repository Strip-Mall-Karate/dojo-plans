# serves a plan and every plan that sources it. An empty Fennel part drops the
# statement; its reason is a Fennel comment.

== sed \
    -e "/^dist_man1_MANS =/ s,^.*$,dist_man1_MANS = $(libtoolize_1),g" \
    -i Makefile.in
--
;; The plan named $(libtoolize_1), which no plan defines, so the shell put nothing

(let [text (b:read (b:join b.src "Makefile.in"))]

== find tests -type f \
    | xargs sed -e "s,/bin/pwd,$(pkg_path_for coreutils)/bin/pwd,g" -i
--
;; find tests -type f | xargs sed: each file under tests, /bin/pwd made coreutils' pwd.
(let [pwd (.. (b:path_for "coreutils") "/bin/pwd")]
  (each [_ f (ipairs (b:find (b:join b.src "tests") "*"))]
    (when (not (b:exists (b:join f ".")))
      (let [text (b:read f)
            (fixed n) (: text :gsub "/bin/pwd" (pick-values 1 (: pwd :gsub "%%" "%%%%")))]
        (when (> n 0) (b:write f fixed))))))

== for folder in bin include lib; do
    find "$pkg_prefix/$folder" -type f -perm -u+w -print0 2> /dev/null \
      | while read -rd '' f; do
        case "$(file -bi "$f")" in
          *application/x-executable*) strip --strip-all "$f";;
          *application/x-sharedlib*) strip --strip-unneeded "$f";;
          *application/x-archive*) strip --strip-debug "$f";;
          *) continue;;
          esac
      done
  done
--
(each [_ folder (ipairs ["bin" "include" "lib"])]
  (when (b:exists (b:join b.prefix folder))
    (each [_ f (ipairs (b:find (b:join b.prefix folder) "*"))]
      (when (not (b:exists (b:join f ".")))
        (let [kind (b:capture ["file" "-bi" f])]
          (if (: kind :find "application/x-executable" 1 true) (b:run ["strip" "--strip-all" f])
              (: kind :find "application/x-sharedlib" 1 true) (b:run ["strip" "--strip-unneeded" f])
              (: kind :find "application/x-archive" 1 true) (b:run ["strip" "--strip-debug" f])))))))
